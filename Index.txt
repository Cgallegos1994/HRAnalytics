			<p>No se encontraron informes que coincidan con tu b&uacute;squeda.</p>

			<p>Intenta con otros t&eacute;rminos o usa los filtros disponibles.</p>
		</div>
	</div>
</div>
<script>
	document.addEventListener('DOMContentLoaded', () =>
	{
		const searchInput = document.getElementById('hr_searchInput');
		const filterButtons = document.querySelectorAll('.hr_filter-btn');
		const cards = document.querySelectorAll('.hr_card');
		const noResults = document.querySelector('.hr_no-results');

		let currentFilter = 'all';

		function filterCards()
		{
			const searchTerm = searchInput.value.toLowerCase().trim();
			let visibleCount = 0;

			cards.forEach(card =>
			{
				const tags = card.dataset.tags.toLowerCase();
				const title = card.querySelector('h2').textContent.toLowerCase();
				const description = card.querySelector('p').textContent.toLowerCase();
				const status = card.dataset.status;

				// --- PASO 1: Comprobar si la tarjeta coincide con el FILTRO del botón activo ---
				let matchesFilter = false;
				if (currentFilter === 'all')
				{
					matchesFilter = true;
				}
				else if (currentFilter === 'disponible' || currentFilter === 'desarrollo')
				{
					if (status === currentFilter)
					{
						matchesFilter = true;
					}
				}
				else
				{
					if (tags.includes(currentFilter))
					{
						matchesFilter = true;
					}
				}

				// --- PASO 2: Comprobar si la tarjeta coincide con el TÉRMINO DE BÚSQUEDA ---
				const cardTextContent = `${title} ${description} ${tags}`;
				const matchesSearch = searchTerm === '' || cardTextContent.includes(searchTerm);

				// --- PASO 3: Decisión final ---
				if (matchesFilter && matchesSearch)
				{
					card.classList.remove('hr_hidden');
					visibleCount++;
				}
				else
				{
					card.classList.add('hr_hidden');
				}
			});

			if (visibleCount === 0)
			{
				noResults.classList.remove('hr_hidden');
			}
			else
			{
				noResults.classList.add('hr_hidden');
			}
		}

		searchInput.addEventListener('input', filterCards);

		filterButtons.forEach(button =>
		{
			button.addEventListener('click', () =>
			{
				filterButtons.forEach(btn => btn.classList.remove('hr_active'));
				button.classList.add('hr_active');

				currentFilter = button.dataset.filter;
				filterCards();
			});
		});

		document.addEventListener('keydown', (e) =>
		{
			if (e.key === '/' && document.activeElement !== searchInput)
			{
				e.preventDefault();
				searchInput.focus();
			}

			if (e.key === 'Escape' && document.activeElement === searchInput)
			{
				searchInput.blur();
			}
		});

		cards.forEach((card, index) =>
		{
			card.style.animationDelay = `${index * 0.05}s`;
		});

		filterCards();
	});

</script>
